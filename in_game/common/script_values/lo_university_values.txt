# Literacy overhaul - university places per area (v0.48, 2026-11-23). Scope: AREA.
# One university place per 400 development of the whole area, at least 1, at most 3. The cap grows with the area: in
# 1500 the run-5 Italian areas had ~650 development (1 place each, Tuscany 954 -> 2), the big German areas 1500-1800
# (3); by 1750 Germany has room for ~55, Italy ~20, France ~30, while thinly settled Russian, Lithuanian and Swedish
# areas (600-800) keep one place. Replaces the fixed "2 per area, 3 in the German lands" of v0.43-v0.47.
# Used by the university allow and lo_university_site (tools/regen_from_base.ps1, $UNI_AREA_CAP).

lo_area_university_places = {
	every_location_in_area = {
		add = development
	}
	divide = 400
	floor = yes
	min = 1
	max = 3
}

lo_area_university_count = {
	every_location_in_area = {
		limit = { has_building = building_type:university }
		add = 1
	}
}

# > 0: the area has room for one more university
lo_area_university_room = {
	value = lo_area_university_places
	subtract = lo_area_university_count
}

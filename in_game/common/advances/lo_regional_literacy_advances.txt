# Literacy overhaul - regional literacy catch-up advances (ported from Flavorium Universalis Balance,
# cc_literacy_advances.txt; keys renamed cc_ -> lo_). They partially restore lower-class literacy where a
# strong tradition existed, offsetting the lower global baseline.

# Age 2: Western European mass literacy (follows Innovativeness)
lo_western_literacy_age2 = {
	age = age_2_renaissance
	icon = scholasticism
	for = adm

	requires = innovativeness

	potential = {
		original_capital ?= { sub_continent = sub_continent:western_europe }
	}

	global_peasants_max_literacy = 2
	global_laborers_max_literacy = 2
	global_soldiers_max_literacy = 5
}

# Age 4: Western European mass literacy (parish schools, printed catechisms)
lo_western_literacy_age4 = {

	}

	global_burghers_max_literacy = 5

# Age 4: colonial ecclesiastical literacy (presses, parish schools, missions)
lo_colonial_literacy_age4 = {
	age = age_4_reformation
	icon = diplomatic_range_age_1
	for = adm

	potential = {
		original_capital ?= { continent = continent:europe }
		any_subject = { is_subject_type = colonial_nation }
	}
	allow = {
		any_subject = { is_subject_type = colonial_nation }
	}

	global_burghers_max_literacy = 2
	global_soldiers_max_literacy = 2
}

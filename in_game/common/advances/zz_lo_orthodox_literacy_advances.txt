# ordered schools for clergy candidates in towns. Russian Orthodox clergy literacy therefore rises only in
# age 4 (the age of Ostroh 1576 and the Kyiv-Mohyla college 1632), while the Greeks rely on their older
# tradition (patriarchate tenet + hesychast advance).

	age = age_4_reformation
	icon = church_councils

	potential = {
		religion = religion:orthodox
		culture = { has_culture_group = culture_group:russian_group }	# base culture group (East Slavic Russians); mods that widen it (e.g. RSF: all East Slavs) widen the advance too
	}

	requires = national_rites

	global_clergy_max_literacy = 10

	# v0.25: the AI values clergy literacy at ~0 and Muscovy never took this in a 1337-1740 run.
	# It also gates the AI weight for basic_religious_education (laws/zz_lo_generated_laws.txt).
	ai_weight = { add = 100 }

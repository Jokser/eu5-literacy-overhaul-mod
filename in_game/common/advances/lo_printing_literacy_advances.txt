# Literacy overhaul - mass literacy from print (v0.34, 2026-11-21).
#
# lo_printed_primers: the lower-class part of the printing press (lower +4, growth +0.01/month) was split off
# printing_press_advance (which keeps the elite part: nobles/clergy 5, burghers 10). Cheap primers, catechisms and
# chapbooks reached the townsfolk and villagers. In the Muslim world printing in Arabic script was not adopted until
# Ibrahim Muteferrika's press (Istanbul 1727; Persia c. 1817, Morocco 1860s) -> Muslims only from age 5.
# Half the normal research cost: it is a follow-up of the printing press, not a new idea.
lo_printed_primers = {
	age = age_3_discovery
	icon = printing_press_advance
	research_cost = 0.5

	# v0.41: the Orthodox also only from age 5 - Muscovite printing (Fedorov's Apostol 1564) was liturgical; the first
	# primers (Fedorov's Lviv bukvar 1574, Burtsov's Moscow bukvar 1634) came late and in small editions.
	potential = {
		OR = {
			AND = {
				NOT = { religion.group = religion_group:muslim }
				NOT = { religion = religion:orthodox }
			}
			current_age = age_5_absolutism
			current_age = age_6_revolutions
		}
	}

	requires = printing_press_advance

	global_laborers_max_literacy = 4
	global_soldiers_max_literacy = 4
	global_peasants_max_literacy = 4
	global_monthly_literacy = 0.01

	ai_weight = { add = 50 }	# the AI values literacy at ~0
}

# lo_church_reading_examinations: second stage of Lutheran mass literacy. The religion itself gives the lower
# classes +4 from the Reformation (catechism, school church orders); the household reading examinations of the
# Swedish church law of 1686 (husforhor) and the Pietist schools (Francke's Halle) came in the age of absolutism.
lo_church_reading_examinations = {
	age = age_5_absolutism
	icon = scholasticism
	research_cost = 0.5

	potential = {
		religion = religion:lutheran
	}

	requires = sanitation_advance

	global_laborers_max_literacy = 4
	global_soldiers_max_literacy = 4
	global_peasants_max_literacy = 4

	ai_weight = { add = 200 }	# v0.41: 50 -> 200 (in 1650 ~20 small Lutheran states had it, but not Sweden, Brunswick, Bohemia, Brandenburg)
}
